cmake_minimum_required(VERSION 3.21)
project(UploadedProgram LANGUAGES C CXX)
if(NOT IS_DIRECTORY "${INPUT_SOURCE}")
  message(FATAL_ERROR "Source directory does not exist: ${INPUT_SOURCE}")
endif()
if(NOT OUTPUT_NAME MATCHES "^[A-Za-z0-9_-]+$")
  message(FATAL_ERROR "Executable name must contain only letters, digits, underscore, or hyphen")
endif()
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "${INPUT_SOURCE}/*.c" "${INPUT_SOURCE}/*.cc" "${INPUT_SOURCE}/*.cpp" "${INPUT_SOURCE}/*.cxx")
list(FILTER SOURCES EXCLUDE REGEX "/(build[^/]*|out|dist|vendor|node_modules|\\.git|CMakeFiles)/")
if(NOT SOURCES)
  message(FATAL_ERROR "No C or C++ source files found in ${INPUT_SOURCE}")
endif()
file(GLOB_RECURSE HEADERS CONFIGURE_DEPENDS "${INPUT_SOURCE}/*.h" "${INPUT_SOURCE}/*.hpp" "${INPUT_SOURCE}/*.hh")
list(FILTER HEADERS EXCLUDE REGEX "/(build[^/]*|out|dist|vendor|node_modules|\\.git|CMakeFiles)/")
set(INCLUDE_DIRS "${INPUT_SOURCE}")
foreach(HEADER IN LISTS HEADERS)
  get_filename_component(HEADER_DIR "${HEADER}" DIRECTORY)
  list(APPEND INCLUDE_DIRS "${HEADER_DIR}")
endforeach()
list(REMOVE_DUPLICATES INCLUDE_DIRS)
add_executable(uploaded_program ${SOURCES})
set_target_properties(uploaded_program PROPERTIES OUTPUT_NAME "${OUTPUT_NAME}" C_STANDARD 17 C_STANDARD_REQUIRED YES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_include_directories(uploaded_program PRIVATE ${INCLUDE_DIRS})
if(MSVC)
  target_compile_options(uploaded_program PRIVATE /W4)
endif()
